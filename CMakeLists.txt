cmake_minimum_required(VERSION 3.18)
project(PPKR LANGUAGES CXX)
include(CTest)
option(PPKR_BUILD_BENCHMARKS "Build OPRF microbenchmarks" ON)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium REQUIRED)
file(STRINGS "${SODIUM_INCLUDE_DIR}/sodium/version.h" SODIUM_VERSION_LINE
     REGEX "^#define SODIUM_VERSION_STRING")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" SODIUM_VERSION "${SODIUM_VERSION_LINE}")
if(NOT SODIUM_VERSION OR SODIUM_VERSION VERSION_LESS "1.0.18")
  message(FATAL_ERROR "libsodium >= 1.0.18 required")
endif()
message(STATUS "libsodium ${SODIUM_VERSION}")
add_library(ppkr_sodium UNKNOWN IMPORTED)
set_target_properties(ppkr_sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${SODIUM_INCLUDE_DIR}")
add_library(ppkr_oprf src/crypto/oprf.cpp)
target_include_directories(ppkr_oprf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ppkr_oprf PUBLIC cxx_std_17)
set_target_properties(ppkr_oprf PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(ppkr_oprf PUBLIC ppkr_sodium)
target_compile_options(ppkr_oprf PRIVATE -Wall -Wextra -Wpedantic)
add_library(ppkr_protocol src/crypto/primitives.cpp src/crypto/shamir.cpp src/protocol/core.cpp src/protocol/backup.cpp src/protocol/sas.cpp)
target_link_libraries(ppkr_protocol PUBLIC ppkr_oprf)
target_compile_options(ppkr_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(protocol_demo examples/protocol_demo.cpp)
target_link_libraries(protocol_demo PRIVATE ppkr_protocol)
add_executable(lifecycle_demo examples/lifecycle_demo.cpp)
target_link_libraries(lifecycle_demo PRIVATE ppkr_protocol)
add_executable(sas_demo examples/sas_demo.cpp)
target_link_libraries(sas_demo PRIVATE ppkr_protocol)
find_package(Git QUIET)
set(PPKR_BUILD_REVISION "unknown")
set(PPKR_BUILD_DIRTY 1)
if(Git_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE PPKR_BUILD_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND ${GIT_EXECUTABLE} diff --quiet HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    RESULT_VARIABLE PPKR_BUILD_DIRTY)
endif()
add_library(ppkr_evaluation src/evaluation/workload.cpp)
target_link_libraries(ppkr_evaluation PUBLIC ppkr_protocol)
target_compile_options(ppkr_evaluation PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(ppkr_evaluation PUBLIC
  PPKR_BUILD_REVISION="${PPKR_BUILD_REVISION}" PPKR_BUILD_DIRTY=${PPKR_BUILD_DIRTY}
  PPKR_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
add_executable(protocol_sizes examples/protocol_sizes.cpp)
target_link_libraries(protocol_sizes PRIVATE ppkr_evaluation)
if(BUILD_TESTING)
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_executable(oprf_tests tests/oprf_tests.cpp)
  target_link_libraries(oprf_tests PRIVATE ppkr_oprf OpenSSL::Crypto)
  add_test(NAME oprf_correctness COMMAND oprf_tests)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE ppkr_protocol)
  add_test(NAME protocol_correctness COMMAND protocol_tests)
  add_executable(backup_tests tests/backup_tests.cpp)
  target_link_libraries(backup_tests PRIVATE ppkr_protocol OpenSSL::Crypto)
  add_test(NAME backup_correctness COMMAND backup_tests)
  add_executable(sas_tests tests/sas_tests.cpp)
  target_link_libraries(sas_tests PRIVATE ppkr_protocol OpenSSL::Crypto)
  add_test(NAME sas_correctness COMMAND sas_tests)
  add_executable(evaluation_tests tests/evaluation_tests.cpp)
  target_link_libraries(evaluation_tests PRIVATE ppkr_evaluation)
  add_test(NAME evaluation_correctness COMMAND evaluation_tests)
endif()
if(PPKR_BUILD_BENCHMARKS)
  find_package(benchmark 1.8.3 CONFIG REQUIRED)
  add_executable(oprf_bench benchmarks/oprf_bench.cpp)
  target_link_libraries(oprf_bench PRIVATE ppkr_oprf benchmark::benchmark)
  add_executable(protocol_bench benchmarks/protocol_bench.cpp)
  target_link_libraries(protocol_bench PRIVATE ppkr_evaluation benchmark::benchmark)
  target_compile_definitions(protocol_bench PRIVATE PPKR_BENCHMARK_VERSION="${benchmark_VERSION}")
  target_compile_definitions(oprf_bench PRIVATE PPKR_BENCHMARK_VERSION="${benchmark_VERSION}")
endif()
