cmake_minimum_required(VERSION 3.18)
project(PPKR LANGUAGES CXX)
include(CTest)
option(PPKR_BUILD_BENCHMARKS "Build OPRF microbenchmarks" ON)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium REQUIRED)
file(STRINGS "${SODIUM_INCLUDE_DIR}/sodium/version.h" SODIUM_VERSION_LINE
     REGEX "^#define SODIUM_VERSION_STRING")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" SODIUM_VERSION "${SODIUM_VERSION_LINE}")
if(NOT SODIUM_VERSION OR SODIUM_VERSION VERSION_LESS "1.0.18")
  message(FATAL_ERROR "libsodium >= 1.0.18 required")
endif()
message(STATUS "libsodium ${SODIUM_VERSION}")
add_library(ppkr_sodium UNKNOWN IMPORTED)
set_target_properties(ppkr_sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${SODIUM_INCLUDE_DIR}")
add_library(ppkr_oprf src/crypto/oprf.cpp)
target_include_directories(ppkr_oprf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ppkr_oprf PUBLIC cxx_std_17)
set_target_properties(ppkr_oprf PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(ppkr_oprf PUBLIC ppkr_sodium)
target_compile_options(ppkr_oprf PRIVATE -Wall -Wextra -Wpedantic)
add_library(ppkr_protocol src/crypto/primitives.cpp src/protocol/core.cpp)
target_link_libraries(ppkr_protocol PUBLIC ppkr_oprf)
target_compile_options(ppkr_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(protocol_demo examples/protocol_demo.cpp)
target_link_libraries(protocol_demo PRIVATE ppkr_protocol)
if(BUILD_TESTING)
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_executable(oprf_tests tests/oprf_tests.cpp)
  target_link_libraries(oprf_tests PRIVATE ppkr_oprf OpenSSL::Crypto)
  add_test(NAME oprf_correctness COMMAND oprf_tests)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE ppkr_protocol)
  add_test(NAME protocol_correctness COMMAND protocol_tests)
endif()
if(PPKR_BUILD_BENCHMARKS)
  find_package(benchmark 1.8.3 CONFIG REQUIRED)
  add_executable(oprf_bench benchmarks/oprf_bench.cpp)
  target_link_libraries(oprf_bench PRIVATE ppkr_oprf benchmark::benchmark)
endif()
